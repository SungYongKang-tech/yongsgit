======================================================================
[장 종료 후 분석파일 만들기]
맨 위 DATE만 바꾸면 됨.

예전 sy-quant-core / kiwwm-server 두 로그를 나눠 보지 않고,
현재 단일 PM2 sy-quant-kr-server 로그를 전략별로 분리해서 저장.

ChatGPT 분석 시 아래 5개를 같이 주는 것이 좋음.
1) CORE/VOLUME
2) OPEN/HOT
3) WAVE
4) FAST
5) MASTER
======================================================================

DATE=2026-08-31
DAY=${DATE//-/}

KR_DIR=/home/ubuntu/sy-quant-kr
KR_OUT=/home/ubuntu/.pm2/logs/sy-quant-kr-server-out.log
KR_ERR=/home/ubuntu/.pm2/logs/sy-quant-kr-server-error.log

echo "분석일: $DATE"

# ============================================================
# 1. CORE / VOLUME
# ============================================================
{
  echo "===== CORE / VOLUME : KR SERVER OUT ====="
  grep "${DATE}T" "$KR_OUT" 2>/dev/null \
    | grep -Ei 'CORE|VOLUME|MASTER|PORTFOLIO|매수|매도|보유|차단|후보|재평가|점수|스위칭' \
    || true

  echo ""
  echo "===== CORE / VOLUME : KR SERVER ERROR ====="
  grep "${DATE}T" "$KR_ERR" 2>/dev/null \
    | grep -Ei 'CORE|VOLUME|MASTER|PORTFOLIO|error|fail|timeout|시간초과|오류|실패' \
    || true

  echo ""
  echo "===== CORE / VOLUME : MASTER SUMMARY ====="
  curl -fsS http://127.0.0.1:3000/api/portfolio-summary \
    | python3 -c '
import sys,json
d=json.load(sys.stdin)
s=d.get("strategies") or {}
print(json.dumps({"CORE":s.get("CORE") or {},"VOLUME":s.get("VOLUME") or {}}, ensure_ascii=False, indent=2))
' 2>/dev/null || echo "CORE/VOLUME MASTER 요약 조회 실패"

} > ~/syquant-kr-core-volume-${DAY}-analysis.txt

# ============================================================
# 2. OPEN / HOT
# HOT은 OPEN 후보탐색 보조자료
# ============================================================
{
  echo "===== OPEN / HOT : KR SERVER OUT ====="
  grep "${DATE}T" "$KR_OUT" 2>/dev/null \
    | grep -Ei 'OPEN|HOT|MASTER|PORTFOLIO|매수|매도|차단|후보|지속|관찰|시장' \
    || true

  echo ""
  echo "===== OPEN / HOT : KR SERVER ERROR ====="
  grep "${DATE}T" "$KR_ERR" 2>/dev/null \
    | grep -Ei 'OPEN|HOT|MASTER|PORTFOLIO|error|fail|timeout|시간초과|API|오류|실패' \
    || true

  echo ""
  echo "===== OPEN LEARNING SUMMARY ====="
  curl -fsS http://127.0.0.1:3000/api/open-learning-summary \
    | python3 -m json.tool 2>/dev/null \
    || echo "OPEN learning summary 조회 실패"

  echo ""
  echo "===== OPEN MASTER SUMMARY ====="
  curl -fsS http://127.0.0.1:3000/api/portfolio-summary \
    | python3 -c '
import sys,json
d=json.load(sys.stdin)
print(json.dumps((d.get("strategies") or {}).get("OPEN") or {}, ensure_ascii=False, indent=2))
' 2>/dev/null || echo "OPEN MASTER 요약 조회 실패"

} > ~/syquant-kr-open-${DAY}-analysis.txt

# ============================================================
# 3. WAVE
# ============================================================
{
  echo "===== WAVE : KR SERVER OUT ====="
  grep "${DATE}T" "$KR_OUT" 2>/dev/null \
    | grep -Ei 'WAVE|READY|TRIGGER|HOLD|PROTECT|REBOUND|COOLDOWN|전일급등|MASTER|PORTFOLIO|매수|매도' \
    || true

  echo ""
  echo "===== WAVE : KR SERVER ERROR ====="
  grep "${DATE}T" "$KR_ERR" 2>/dev/null \
    | grep -Ei 'WAVE|wave-strategy|MASTER|PORTFOLIO|error|fail|timeout|시간초과|오류|실패' \
    || true

  echo ""
  echo "===== WAVE LOCAL STATE ====="
  if [ -f "$KR_DIR/paper-state-wave.json" ]; then
    cat "$KR_DIR/paper-state-wave.json"
  else
    echo "paper-state-wave.json 없음 또는 현재 구조에서 별도 상태파일 미사용"
  fi

  echo ""
  echo "===== MASTER WAVE SUMMARY ====="
  curl -fsS http://127.0.0.1:3000/api/portfolio-summary \
    | python3 -c '
import sys,json
d=json.load(sys.stdin)
print(json.dumps((d.get("strategies") or {}).get("WAVE") or {}, ensure_ascii=False, indent=2))
' 2>/dev/null || echo "MASTER WAVE 요약 조회 실패"

} > ~/syquant-kr-wave-${DAY}-analysis.txt

# ============================================================
# 4. FAST
# ============================================================
{
  echo "===== FAST : KR SERVER OUT ====="
  grep "${DATE}T" "$KR_OUT" 2>/dev/null \
    | grep -Ei 'FAST|MASTER|PORTFOLIO|매수|매도|후보|점수|트리거|관찰|진입|청산|손절|익절|보유' \
    || true

  echo ""
  echo "===== FAST : KR SERVER ERROR ====="
  grep "${DATE}T" "$KR_ERR" 2>/dev/null \
    | grep -Ei 'FAST|MASTER|PORTFOLIO|error|fail|timeout|시간초과|API|오류|실패' \
    || true

  echo ""
  echo "===== FAST LOCAL STATE ====="
  if [ -f "$KR_DIR/paper-state-fast.json" ]; then
    cat "$KR_DIR/paper-state-fast.json"
  else
    echo "paper-state-fast.json 없음 또는 현재 구조에서 별도 상태파일 미사용"
  fi

  echo ""
  echo "===== MASTER FAST SUMMARY ====="
  curl -fsS http://127.0.0.1:3000/api/portfolio-summary \
    | python3 -c '
import sys,json
d=json.load(sys.stdin)
print(json.dumps((d.get("strategies") or {}).get("FAST") or {}, ensure_ascii=False, indent=2))
' 2>/dev/null || echo "MASTER FAST 요약 조회 실패"

} > ~/syquant-kr-fast-${DAY}-analysis.txt

# ============================================================
# 5. MASTER 전체 계좌
# ============================================================
{
  echo "===== MASTER PORTFOLIO SUMMARY ====="
  curl -fsS http://127.0.0.1:3000/api/portfolio-summary \
    | python3 -m json.tool \
    || echo "MASTER portfolio 조회 실패"

  echo ""
  echo "===== PERFORMANCE SUMMARY ====="
  curl -fsS http://127.0.0.1:3000/api/performance-summary \
    | python3 -m json.tool 2>/dev/null \
    || echo "performance-summary 조회 실패"

  echo ""
  echo "===== PAPER STATE FILE LIST ====="
  find "$KR_DIR" -maxdepth 1 -type f -name 'paper-state*.json' -printf '%f\n' 2>/dev/null | sort

  echo ""
  echo "===== MASTER FINANCIAL STATE ====="

  STATE=""
  for p in "$KR_DIR/paper-state-core.json" "$KR_DIR/paper-state.json"; do
    if [ -f "$p" ]; then STATE="$p"; break; fi
  done

  if [ -n "$STATE" ]; then
    echo "사용 상태파일: $STATE"
    STATE="$STATE" python3 - <<'PY'
import os,json
p=os.environ["STATE"]
with open(p, encoding="utf-8") as f:
    d=json.load(f)

def n(v):
    try: return int(float(v or 0))
    except: return 0

print("초기자산:", f'{n(d.get("initialCapital")):,}원')
print("현금:", f'{n(d.get("totalCash")):,}원')
holdings=d.get("holdings") or []
logs=d.get("tradeLogs") or []
print("보유종목:", len(holdings))
print("거래로그:", len(logs))
print("\n----- 현재 보유 -----")
for x in holdings:
    print(x.get("strategyGroup") or x.get("strategy"), x.get("name"), x.get("code"), "수량", x.get("qty"), "매수가", x.get("buyPrice"), "현재가", x.get("currentPrice"))
print("\n----- 최근 거래 50건 -----")
for x in logs[-50:]:
    print(x.get("time") or x.get("date"), x.get("strategyGroup") or x.get("strategy"), x.get("type"), x.get("name"), x.get("code"), "손익", x.get("profit"), "수익률", x.get("profitRate"))
PY
  else
    echo "MASTER 상태파일을 자동으로 찾지 못함"
  fi

  echo ""
  echo "===== RECENT KR SERVER ERRORS ====="
  tail -n 100 "$KR_ERR" 2>/dev/null || true

} > ~/syquant-kr-master-${DAY}-analysis.txt

echo ""
echo "===== 생성 완료 ====="
ls -lh \
  ~/syquant-kr-core-volume-${DAY}-analysis.txt \
  ~/syquant-kr-open-${DAY}-analysis.txt \
  ~/syquant-kr-wave-${DAY}-analysis.txt \
  ~/syquant-kr-fast-${DAY}-analysis.txt \
  ~/syquant-kr-master-${DAY}-analysis.txt

======================================================================
[전체 리셋 - 주의]
======================================================================

예전 메모의 ~/kiwoom-server 경로는 현재 틀림.
리셋은 데이터 삭제 작업이므로 현재 server.js에 API가 실제 존재하는지 먼저 확인.

cd /home/ubuntu/sy-quant-kr
grep -n 'api/paper-state/reset' server.js

위 grep에서 reset API가 확인될 때만:
curl -X POST http://127.0.0.1:3000/api/paper-state/reset

확인:
curl -s http://127.0.0.1:3000/api/portfolio-summary | python3 -m json.tool

※ 리셋 전 paper-state*.json 백업 권장.
※ 장중 리셋 금지.

cd /home/ubuntu/sy-quant-kr
STATE=""
for p in paper-state-core.json paper-state.json; do
  [ -f "$p" ] && STATE="$p" && break
done

echo "STATE=$STATE"

if [ -n "$STATE" ]; then
  jq --arg today "$(date +%F)" '
    .tradeLogs[]
    | select((.date // "") | contains($today))
    | {time,name,type,strategyGroup,strategyName,discoverScore,buyPrice,sellPrice,profit,profitRate,changeRate,volume}
  ' "$STATE"
fi
