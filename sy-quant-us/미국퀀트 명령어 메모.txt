======================================================================
[장 종료 후 US 분석파일 만들기]
맨 위 DATE만 미국 거래일로 변경.
======================================================================

DATE=2026-08-31
DAY=${DATE//-/}

US_DIR=/home/ubuntu/sy-quant-us
US_OUT=/home/ubuntu/.pm2/logs/sy-quant-us-server-out.log
US_ERR=/home/ubuntu/.pm2/logs/sy-quant-us-server-error.log

echo "미국 분석일: $DATE"

# ============================================================
# 1. US-CORE 관찰 / 후보 / 오류
# ============================================================
{
  echo "===== US-CORE SERVER LOG ====="
  grep "${DATE}T" "$US_OUT" 2>/dev/null \
    | grep -E 'US-CORE|READY|WATCH|OBSERVE|가상진입|가상추적|이력|일일요약|데이터보정|오류' \
    || true

  echo ""
  echo "===== US-CORE ERROR LOG ====="
  grep "${DATE}T" "$US_ERR" 2>/dev/null \
    | grep -Ei 'US-CORE|CORE|error|fail|timeout|오류|실패' \
    || true

  echo ""
  echo "===== US-CORE STATUS ====="
  curl -fsS http://127.0.0.1:3001/api/us-core/status \
    | python3 -m json.tool \
    || echo "US-CORE status 조회 실패"

  echo ""
  echo "===== US-CORE HISTORY ====="
  curl -fsS "http://127.0.0.1:3001/api/us-core/history?date=${DATE}" \
    | python3 -m json.tool \
    || echo "US-CORE history 조회 실패"

} > ~/syquant-us-core-${DAY}-analysis.txt

# ============================================================
# 2. US-CORE 가상진입 / 장마감요약 / 매도법 비교
# ============================================================
{
  echo "===== US-CORE VIRTUAL TRADES ====="
  curl -fsS http://127.0.0.1:3001/api/us-core/virtual-trades \
    | python3 -m json.tool \
    || echo "virtual trades 조회 실패"

  echo ""
  echo "===== US-CORE DAILY SUMMARY ====="
  curl -fsS "http://127.0.0.1:3001/api/us-core/daily-summary?date=${DATE}" \
    | python3 -m json.tool \
    || echo "daily summary 조회 실패"

  echo ""
  echo "===== US-CORE DAILY SUMMARY TXT ====="
  TXT="$US_DIR/us-core-reports/us-core-summary-${DAY}.txt"
  if [ -f "$TXT" ]; then
    cat "$TXT"
  else
    echo "장마감 FINAL TXT 아직 없음: $TXT"
  fi

  echo ""
  echo "===== EXIT SIMULATION ====="
  node "$US_DIR/us-core-exit-simulator.js" 2>&1

} > ~/syquant-us-core-result-${DAY}-analysis.txt

# ============================================================
# 3. US MASTER / 안전상태 / 서버상태
# ============================================================
{
  echo "===== US SERVER STATUS ====="
  curl -fsS http://127.0.0.1:3001/api/status \
    | python3 -m json.tool \
    || echo "US status 조회 실패"

  echo ""
  echo "===== US PORTFOLIO ====="
  curl -fsS http://127.0.0.1:3001/api/portfolio-summary \
    | python3 -m json.tool \
    || echo "US portfolio 조회 실패"

  echo ""
  echo "===== US STRATEGY SETTINGS ====="
  curl -fsS http://127.0.0.1:3001/api/strategy-settings \
    | python3 -m json.tool \
    || echo "US strategy settings 조회 실패"

  echo ""
  echo "===== CORE BUY CHECK ====="
  curl -fsS http://127.0.0.1:3001/api/strategy-buy-check/CORE \
    | python3 -m json.tool \
    || echo "CORE buy-check 조회 실패"

  echo ""
  echo "===== HISTORY RECORDER STATUS ====="
  curl -fsS http://127.0.0.1:3001/api/us-core/history-status \
    | python3 -m json.tool \
    || echo "history status 조회 실패"

  echo ""
  echo "===== DAILY SUMMARY STATUS ====="
  curl -fsS http://127.0.0.1:3001/api/us-core/daily-summary-status \
    | python3 -m json.tool \
    || echo "daily summary status 조회 실패"

  echo ""
  echo "===== RECENT US ERRORS ====="
  tail -n 100 "$US_ERR" 2>/dev/null || true

} > ~/syquant-us-master-${DAY}-analysis.txt

echo ""
echo "===== 생성 완료 ====="
ls -lh \
  ~/syquant-us-core-${DAY}-analysis.txt \
  ~/syquant-us-core-result-${DAY}-analysis.txt \
  ~/syquant-us-master-${DAY}-analysis.txt
